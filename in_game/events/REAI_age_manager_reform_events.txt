# 4. Çağ Better reform verme eventi
reai_age_manager_events.1 = {
	type = country_event
	title = reai_age_manager_events.1.t
	desc = reai_age_manager_events.1.d
	hidden = yes
	trigger = {
		is_ai = yes
		government_reform:statute_of_monopolies = {
			is_allowed_for = root
		}
		government_reform:statute_of_monopolies = {
			is_implementable_in = root
		}
		NOT = {
			has_reform = government_reform:statute_of_monopolies
		}
		num_open_reform_slots >= 1
		# AI'ın Merkantilizm tarafında (sol eksen) %50'den fazla ilerlemiş olması şartı
		societal_value:mercantilism_vs_free_trade < -50
	}
	immediate = {
	}
	option = {
		name = reai_age_manager_events.1.a
		add_reform = government_reform:statute_of_monopolies
	}
}
